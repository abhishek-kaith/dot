# Installed on every machine, whatever the desktop.
# One package per word; everything after # is ignored.
# Not listed because archinstall already installs them: base, linux,
# linux-firmware, amd-ucode, networkmanager, pipewire (+ pulse/alsa/jack),
# wireplumber, sudo. mesa comes with the desktop.

# ── shell ──
zsh zsh-completions zsh-autosuggestions zsh-syntax-highlighting starship
tmux neovim tree-sitter-cli mise

# ── git + dev ──
git github-cli git-delta lazygit base-devel openssh sshfs

# ── modern cli ──
ripgrep fd fzf zoxide bat eza dust btop jq go-yq xh tealdeer
man-db man-pages
unzip zip 7zip unrar rsync wget wl-clipboard
powertop                      # measure battery draw: sudo powertop

# ── network + dns ──
bind                          # dig, host, nslookup
doggo                         # modern dig (DoH/DoT aware)
mtr gping                     # live traceroute / ping graph
nmap openbsd-netcat           # port scan / nc
iperf3 tcpdump ethtool        # bandwidth test / packet capture / link info
bandwhich                     # live per-process bandwidth
whois wireguard-tools

# ── containers (rootless podman; `docker` works too) ──
podman podman-docker          # podman + a `docker` command that calls it
docker-compose                # compose; talks to podman.socket (DOCKER_HOST)
lazydocker dive               # TUI for containers / inspect image layers

# ── apps ──
ghostty                       # terminal (GPU)
chromium                      # browser (VA-API on AMD since 143)
bitwarden bitwarden-cli       # passwords + SSH agent (desktop app), `bw` CLI
mpv                           # video (GPU)
libreoffice-fresh             # docx / xlsx / pptx
obs-studio                    # screen recording / streaming (NVENC on NVIDIA, VA-API on AMD)
telegram-desktop              # Telegram (Qt, follows light/dark)
shotwell                      # photo gallery + phone-style edits: crop, straighten, enhance
pinta                         # quick drawing / markup / resize (Paint.NET-like)
scrcpy                        # show + control the Android phone on the laptop (USB or Wi-Fi)
qbittorrent                   # torrents (Qt, follows light/dark)

# ── graphics + media ──
vulkan-radeon                 # AMD Vulkan; without it Vulkan apps wake the NVIDIA GPU
gst-libav gst-plugins-good    # codecs for GStreamer apps
noise-suppression-for-voice   # mic denoise (~/.config/pipewire/pipewire.conf.d)

# ── system ──
power-profiles-daemon         # power modes; KDE + GNOME switchers use it
rtkit                         # realtime priority for PipeWire (no audio crackles under load)
wireless-regdb                # Wi-Fi regulatory database (kernel: "regulatory.db failed")
fwupd                         # firmware updates: fwupdmgr update
ufw                           # firewall
reflector pacman-contrib      # mirror refresh + paccache
zram-generator
bluez bluez-utils
cups                          # printing
flatpak
fuse2                         # AppImage support
xdg-desktop-portal-gtk        # GTK apps and Flatpaks: file dialogs, cursor theme

# ── fonts ──
noto-fonts noto-fonts-cjk noto-fonts-emoji
hunspell-en_us                # English spell check (KDE apps, LibreOffice)
ttf-jetbrains-mono-nerd       # terminal + editor font (icons included)
